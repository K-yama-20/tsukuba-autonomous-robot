cmake_minimum_required(VERSION 3.16)
project(gouda_sim)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(virtual_vehicle src/virtual_vehicle.cpp)
target_compile_features(virtual_vehicle PUBLIC cxx_std_17)
target_include_directories(virtual_vehicle PRIVATE ../firmware/gouda_dualsense_usb/include)
ament_target_dependencies(virtual_vehicle rclcpp std_msgs geometry_msgs std_srvs)
target_link_libraries(virtual_vehicle nlohmann_json::nlohmann_json)
install(TARGETS virtual_vehicle DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/sensor_adapter scripts/scenarios DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch worlds config DESTINATION share/${PROJECT_NAME})
ament_package()
