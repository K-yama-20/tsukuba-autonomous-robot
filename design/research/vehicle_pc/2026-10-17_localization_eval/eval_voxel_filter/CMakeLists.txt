cmake_minimum_required(VERSION 3.8)
project(eval_voxel_filter)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters)
add_executable(voxel_filter src/voxel_filter.cpp)
target_include_directories(voxel_filter PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(voxel_filter ${PCL_LIBRARIES})
ament_target_dependencies(voxel_filter rclcpp sensor_msgs pcl_conversions)
install(TARGETS voxel_filter DESTINATION lib/${PROJECT_NAME})
ament_package()
